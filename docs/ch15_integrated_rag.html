10. 통합 컨트롤러

코드

package com.example.rag;
import java.util.List;
import java.util.Map;
import org.springframework.ai.chat.client.ChatClient;
import org.springframework.ai.document.Document;
import org.springframework.ai.vectorstore.VectorStore;
import org.springframework.ai.vectorstore.SearchRequest;
import org.springframework.web.bind.annotation.*;

@RestController
@RequestMapping("/api")
public class RagController {
    private final VectorStore store;
    private final ChatClient chat;

    public RagController(VectorStore store, ChatClient.Builder builder) {
        this.store = store;
        this.chat = builder.build();
    }

    // 교육용 고정 문서: 실제 운영 시 인증과 적재 검증 필수
    @PostMapping("/seed")
    public Map<String, Object> seed() {
        store.add(List.of(
            new Document("UART는 비동기 직렬 통신 방식이며 TX와 RX를 사용합니다.",
                Map.of("source", "uart-note", "access", "public")),
            new Document("PWM은 듀티비를 조절해 평균 출력 전력을 제어합니다.",
                Map.of("source", "pwm-note", "access", "public"))
        ));
        return Map.of("inserted", 2);
    }

    @GetMapping("/ask")
    public Map<String, Object> ask(@RequestParam String q) {
        if (q.isBlank() || q.length() > 300)
            throw new IllegalArgumentException("질문 길이는 1~300자입니다.");
        List<Document> hits = store.similaritySearch(
            SearchRequest.builder().query(q).topK(3).build());
        if (hits == null || hits.isEmpty())
            return Map.of("answer", "검색된 근거가 없습니다.", "sources", List.of());
        StringBuilder context = new StringBuilder();
        for (Document d : hits)
            context.append("[").append(d.getMetadata().get("source"))
                   .append("] ").append(d.getText()).append("\n");
        String answer = chat.prompt()
            .user("다음 근거만 이용해 한국어로 답하세요. 근거가 부족하면 모른다고 하세요.\n"
                + context + "\n질문: " + q)
            .call().content();
        return Map.of("answer", answer == null ? "" : answer,
            "sources", hits.stream().map(d ->
                String.valueOf(d.getMetadata().get("source"))).distinct().toList());
    }
}

예상 출력

/api/seed 및 /api/ask 엔드포인트가 등록됩니다.

코드 풀이 — 이 코드가 무엇을 하는가

검색된 문서만 프롬프트에 포함하고 출처 목록을 별도로 반환합니다. 모델 답변이 출처를 실제로 충실히 따르는지는 추가 평가가 필요합니다.
