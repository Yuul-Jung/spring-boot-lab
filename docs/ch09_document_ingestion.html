05청크 설계 · 규칙과 메타데이터

필드역할
source_id원본 파일 식별자
title원본 첫 줄 제목
chunk_index문서 내 청크 순번(0부터)
content임베딩 및 검색할 본문
content_hash본문 SHA-256 해시
embeddingbge-m3의 1024차원 벡터

이 실습은 빈 줄로 구분된 문단을 하나의 청크로 사용한다. 문단이 길어지는 실제 PDF·법령 자료에서는 토큰 기준 분할, 중첩, 표·제목 계층 보존이 추가로 필요하다. 파일명과 청크 순번을 이용한 안정적인 문서 ID로 재실행 시 중복 색인을 방지한다.

06전체 적재 프로그램 · ingest.py

CODE · python · ingest.py
from pathlib import Path
import hashlib
import json
import os
import requests

BASE = Path(__file__).resolve().parent
DATA = BASE / "data"
OLLAMA = os.getenv("OLLAMA_URL", "http://127.0.0.1:11434")
ES = os.getenv("ES_URL", "http://127.0.0.1:9200")
INDEX = "rag_documents_ch09"
MODEL = "bge-m3"
DIM = 1024

session = requests.Session()
session.trust_env = False  # localhost 요청에 외부 프록시가 개입하지 않도록 함


def request_json(method, url, **kwargs):
    response = session.request(method, url, timeout=120, **kwargs)
    response.raise_for_status()
    return response.json()


def chunks_from_file(path):
    text = path.read_text(encoding="utf-8").replace("\r\n", "\n")
    paragraphs = [p.strip() for p in text.split("\n\n") if p.strip()]
    title = paragraphs[0].removeprefix("제목:").strip()
    for index, paragraph in enumerate(paragraphs[1:]):
        yield {
            "source_id": path.name,
            "title": title,
            "chunk_index": index,
            "content": paragraph,
            "content_hash": hashlib.sha256(
                paragraph.encode("utf-8")
            ).hexdigest(),
        }


def embed(text):
    data = request_json(
        "POST", f"{OLLAMA}/api/embed",
        json={"model": MODEL, "input": text}
    )
    vectors = data.get("embeddings", [])
    if len(vectors) != 1 or len(vectors[0]) != DIM:
        raise ValueError("임베딩 개수 또는 차원 오류")
    return vectors[0]


def ensure_index():
    mapping = {
        "mappings": {
            "properties": {
                "source_id": {"type": "keyword"},
                "title": {"type": "text"},
                "chunk_index": {"type": "integer"},
                "content": {"type": "text"},
                "content_hash": {"type": "keyword"},
                "embedding": {
                    "type": "dense_vector",
                    "dims": DIM,
                    "index": True,
                    "similarity": "cosine"
                }
            }
        }
    }
    exists = session.head(f"{ES}/{INDEX}", timeout=10)
    if exists.status_code == 404:
        request_json("PUT", f"{ES}/{INDEX}", json=mapping)
    elif exists.status_code != 200:
        exists.raise_for_status()


def main():
    ensure_index()
    total = 0
    with (BASE / "chunks.jsonl").open("w", encoding="utf-8") as out:
        for path in sorted(DATA.glob("*.txt")):
            for chunk in chunks_from_file(path):
                chunk["embedding"] = embed(chunk["content"])
                doc_id = f'{chunk["source_id"]}:{chunk["chunk_index"]}'
                request_json(
                    "PUT", f"{ES}/{INDEX}/_doc/{doc_id}",
                    json=chunk
                )
                out.write(json.dumps(
                    {k: v for k, v in chunk.items() if k != "embedding"},
                    ensure_ascii=False
                ) + "\n")
                total += 1
    request_json("POST", f"{ES}/{INDEX}/_refresh")
    print(f"적재 완료: {total}개 청크")


if __name__ == "__main__":
    main()
EXPECTED OUTPUT · 예상 결과
적재 완료: 6개 청크
코드 풀이 — 이 코드가 무엇을 하는가

두 문서의 첫 문단은 제목으로 사용하고 나머지 문단 3개씩 총 6개 청크를 적재한다. embed()는 Ollama에 각 청크의 벡터를 요청하고 1024차원인지 검사한다. ensure_index()는 Elasticsearch의 기존 인덱스를 보존하고 없을 때만 생성한다. 재실행은 같은 문서 ID를 덮어쓰지만 삭제된 원본 청크를 자동 정리하지는 않는다.

08검색 검증 · 질문 벡터로 찾기

CODE · python · search.py
from ingest import ES, INDEX, embed, request_json

question = "UART 8N1은 어떤 의미인가?"
vector = embed(question)
query = {
    "size": 3,
    "_source": ["source_id", "title", "chunk_index", "content"],
    "knn": {
        "field": "embedding",
        "query_vector": vector,
        "k": 3,
        "num_candidates": 6
    }
}
result = request_json("POST", f"{ES}/{INDEX}/_search", json=query)
for hit in result["hits"]["hits"]:
    item = hit["_source"]
    print(f'{item["source_id"]} #{item["chunk_index"]}: '
          f'{item["content"][:80]}')
