04 · 전체 실습 코드: domain_rag.py

import requests

ES = "http://127.0.0.1:9200"
OLLAMA = "http://127.0.0.1:11434"
INDEX = "rag_domain_ch12"

docs = [
 {"id":"erp1","title":"구매 요청","kind":"erp","level":"public",
  "dept":"purchase","text":"구매 요청은 부서장의 승인을 받는다."},
 {"id":"erp2","title":"결산 검토","kind":"erp","level":"internal",
  "dept":"finance","text":"결산 검토표와 승인 이력을 기록한다."},
 {"id":"tech1","title":"UART 점검","kind":"technical","level":"public",
  "dept":"engineering","text":"UART 시작 비트와 정지 비트를 확인한다."},
 {"id":"tech2","title":"모터 보호","kind":"technical","level":"internal",
  "dept":"engineering","text":"과전류 보호와 전원 분리를 검토한다."}
]

def embed(text):
    r = requests.post(f"{OLLAMA}/api/embed",
        json={"model":"bge-m3","input":text},timeout=120)
    r.raise_for_status()
    v = r.json()["embeddings"][0]
    if len(v) != 1024:
        raise ValueError("임베딩 차원 오류")
    return v

mapping = {"mappings":{"properties":{
    "title":{"type":"text"},"text":{"type":"text"},
    "kind":{"type":"keyword"},"level":{"type":"keyword"},
    "dept":{"type":"keyword"},
    "vector":{"type":"dense_vector","dims":1024,
              "index":True,"similarity":"cosine"}
}}}
r = requests.put(f"{ES}/{INDEX}",json=mapping,timeout=30)
if r.status_code == 400:
    if r.json().get("error",{}).get("type") != "resource_already_exists_exception":
        r.raise_for_status()
else:
    r.raise_for_status()

for doc in docs:
    body = {k:v for k,v in doc.items() if k != "id"}
    body["vector"] = embed(doc["title"]+" "+doc["text"])
    r = requests.put(f"{ES}/{INDEX}/_doc/{doc['id']}",
        params={"refresh":"true"},json=body,timeout=120)
    r.raise_for_status()
print("색인 완료: 4건")

def search(question,role,kind):
    levels = {"guest":["public"],"staff":["public","internal"]}
    if role not in levels:
        raise ValueError("허용되지 않은 역할")
    filters = [{"terms":{"level":levels[role]}},
               {"term":{"kind":kind}}]
    query = {"size":3,"_source":["title","kind","dept","level"],
      "knn":{"field":"vector","query_vector":embed(question),
             "k":3,"num_candidates":10,
             "filter":{"bool":{"filter":filters}}}}
    r = requests.post(f"{ES}/{INDEX}/_search",
        json=query,timeout=120)
    r.raise_for_status()
    for hit in r.json()["hits"]["hits"]:
        d = hit["_source"]
        print(hit["_id"],d["title"],d["level"])

print("[guest]")
search("구매 결산 승인","guest","erp")
print("[staff]")
search("구매 결산 승인","staff","erp")

코드 풀이: 고정 ID를 사용하므로 반복 실행해도 문서가 중복되지 않는다. Elasticsearch의 kNN 검색에 등급 및 유형 필터를 적용한다. 임베딩 차원은 1024로 검증한다.

연결 실패는 Docker 및 Ollama 상태를 점검한다. 모델 미설치는 ollama pull bge-m3로 해결한다. 인덱스 매핑이 다르면 기존 전용 인덱스를 확인한다. Elasticsearch 직접 접속은 이 교육용 필터를 우회할 수 있으므로 외부에 공개하지 않는다.

07 · Spring AI 연계 및 핵심 정리

인증된 사용자 → 서버 역할 확인 → 필터된 검색 결과 → Spring AI ChatModel → 답변·출처

제10장의 답변 생성에는 권한을 통과한 문서만 전달한다. title·kind·dept를 출처로 표시하고 실제 서비스에서는 서버 측 인증, 문서별 ACL, 감사 로그가 필요하다. 기존 Java 파일은 변경하지 않는다.
